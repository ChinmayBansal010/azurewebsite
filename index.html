Developer SDK

Integrate autonomous intelligence into your production stack with five lines of code.

COPY
# Install: pip install automl-os
import automl_os as os_ai

# Initialize client with Enterprise context
client = os_ai.Client(api_key="AM_OS_X772", cluster="us-east-1")

# Start autonomous search & training
model = client.pipelines.create(
    name="UserChurn_v4",
    strategy="max_accuracy",
    compute=os_ai.Compute.GPU_H100_XL
)

# Automatic deployment to Global Edge
endpoint = model.deploy(target="global_edge_cdn")
print(f"Endpoint Live: {endpoint.url}")
